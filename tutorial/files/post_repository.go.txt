package repository

import (
	"context"

	"example.com/blog/model"
	"example.com/blog/scopes"
)

//go:generate go tool kanna-orm -destination ../postgres

//kanna:repository table=posts
type Post interface {
	Get(ctx context.Context, id int64, s ...scopes.Post) (model.Post, error)
	List(ctx context.Context, s ...scopes.Post) ([]model.Post, error)
	Insert(ctx context.Context, post model.Post) (model.Post, error)
	Update(ctx context.Context, post model.Post) error
	Delete(ctx context.Context, id int64) error
}
